1. A Hierarquia dos 4 Tipos de Estado

Tipo de Estado Onde Reside Ferramenta Recomendada Exemplo de Uso
Server State (Dados Assíncronos) src/features//api/ TanStack Query (React Query) Lista de produtos, dados do carrinho, perfil do usuário.
URL State (Compartilhável) URL Search Params (Browser) nuqs ou useSearchParams Filtros de categoria, ordenação de preço, página atual.
Feature UI State (Síncrono da Feature) src/features//stores/ Zustand Slice ou React Context Drawer do carrinho aberto, etapa do formulário de checkout.
Local Component State (Efêmero) Direto no componente .tsx useState / useReducer Campo de input controlado, hover de dropdown, accordion aberto.

2. Padrão Query Key Factory (TkDodo)

Previne colisões acidentais de cache e garante tipagem estrita na invalidação:

// src/features/cart/api/query-keys.ts
export const cartKeys = {
  all: ['cart'] as const,
  details: () => [...cartKeys.all, 'detail'] as const,
  items: () => [...cartKeys.all, 'items'] as const,
  item: (id: string) => [...cartKeys.items(), id] as const,
};

// Como usar em hooks de leitura:
useQuery({
  queryKey: cartKeys.details(),
  queryFn: getCartDetails,
});

// Como invalidar cirurgicamente em qualquer lugar:
queryClient.invalidateQueries({ queryKey: cartKeys.all });

3. Anatomia de uma Mutation Otimista (Optimistic Update)

// src/features/cart/api/use-add-to-cart.ts
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { cartKeys } from './query-keys';
import { addToCartApi } from './cart-api';
import type { Cart, CartItem } from '../types';

export function useAddToCartMutation() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: addToCartApi,

    // 1. Antes de disparar a requisição: atualização otimista na UI
    onMutate: async (newItem: CartItem) => {
      await queryClient.cancelQueries({ queryKey: cartKeys.details() });

      const previousCart = queryClient.getQueryData<Cart>(cartKeys.details());

      if (previousCart) {
        queryClient.setQueryData<Cart>(cartKeys.details(), {
          ...previousCart,
          items: [...previousCart.items, newItem],
          totalInCents: previousCart.totalInCents + newItem.priceInCents * newItem.quantity,
        });
      }

      return { previousCart }; // Contexto para rollback se falhar
    },

    // 2. Se a requisição HTTP falhar: reverte para o estado anterior
    onError: (err, newItem, context) => {
      if (context?.previousCart) {
        queryClient.setQueryData(cartKeys.details(), context.previousCart);
      }
    },

    // 3. Sempre sincroniza com a verdade final do servidor
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: cartKeys.details() });
    },
  });
}

4.1 Handlers MSW Isolados por Feature

// src/features/cart/testing/handlers.ts
import { http, HttpResponse } from 'msw';
import { mockCartData } from './fixtures';

export const cartHandlers = [
  http.get('/api/cart', () => HttpResponse.json(mockCartData)),
  http.post('/api/cart/coupon', async ({ request }) => {
    const { code } = (await request.json()) as { code: string };
    if (code === 'INVALIDO') {
      return HttpResponse.json({ message: 'Cupom inválido' }, { status: 400 });
    }
    return HttpResponse.json({ ...mockCartData, discountInCents: 1500 });
  }),
];

4.2 Teste de Integração da Feature

// src/features/cart/components/cart-drawer.test.tsx
import { render, screen, waitFor } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { createTestQueryClientWrapper } from '@/testing/test-utils';
import { CartDrawer } from './cart-drawer';

describe('Feature: CartDrawer (Integration)', () => {
  it('deve aplicar cupom e recalcular o total via mutação interceptada pelo MSW', async () => {
    const user = userEvent.setup();
    const { Wrapper } = createTestQueryClientWrapper();

    render(<CartDrawer isOpen={true} onClose={() => {}} />, { wrapper: Wrapper });

    expect(await screen.findByText(/Tênis Esportivo/i)).toBeInTheDocument();

    const input = screen.getByPlaceholderText(/inserir cupom/i);
    await user.type(input, 'PROMO10');
    await user.click(screen.getByRole('button', { name: /aplicar/i }));

    await waitFor(() => {
      expect(screen.getByText(/Desconto/i)).toBeInTheDocument();
    });
  });
});
